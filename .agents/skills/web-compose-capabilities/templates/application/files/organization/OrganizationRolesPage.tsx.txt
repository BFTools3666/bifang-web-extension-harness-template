import { ApiError } from "../../data/api/client/httpClient"
import { ListSurface, ListTable, ListCell } from "../../pagination/ListSurface"
import {
  useTableColumns,
  TableColumnControls,
  TableColumnHeader
} from "../../pagination/TableColumns"
import { useEffect, useState } from "react"
import { useQuery } from "@tanstack/react-query"
import { Badge, Button, Group, Stack, Table, Text, Title } from "@mantine/core"
import { useTranslation } from "react-i18next"
import { ModuleState } from "../application/ModuleState"
import { useManagementScope, useManagementRejection } from "./useManagement"
import { useManagementList } from "./managementList"
import { rolesQuery } from "./managementQueries"
import { RoleDialog, type RoleAction } from "./RoleDialog"
import { ManagementPagination } from "./ManagementParts"
import classes from "./management.module.css"
import "../../i18n/organizationManagement"

/** 组织角色独立分页，系统角色及超出操作者权限的角色只允许查看。 */
export function OrganizationRolesPage() {
  const { t, i18n } = useTranslation(),
    { owner, ownerKey, allow } = useManagementScope()
  const list = useManagementList(ownerKey)
  const tableScope = JSON.stringify([owner?.accountId, owner?.organizationId])
  const columns = useTableColumns(
    "organization-roles",
    tableScope,
    [
      { id: "role", label: t("orgRole"), required: true, width: 240 },
      { id: "permissions", label: t("orgPermissions"), width: 160 },
      { id: "type", label: t("orgType"), width: 160 },
      { id: "actions", label: t("orgActions"), required: true, width: 280 }
    ],
    true
  )
  const [action, setAction] = useState<{ kind: RoleAction; id?: string } | null>(null)
  const query = useQuery({
    ...rolesQuery(ownerKey, owner?.organizationId ?? "", i18n.language, list.state),
    enabled: !!owner
  })
  const denied =
    query.error instanceof ApiError && (query.error.status === 401 || query.error.status === 403)
  const readable = !!owner && !denied
  useEffect(() => {
    if (!readable) setAction(null)
  }, [readable])
  useManagementRejection(query.error, ownerKey)
  useEffect(() => {
    if (query.isSuccess && !query.isFetching && !query.isPlaceholderData)
      list.correct(query.data.total)
  }, [
    query.data,
    query.isFetching,
    query.isSuccess,
    ownerKey,
    list.state.page,
    list.state.pageSize
  ])
  const target = query.data?.items.find((role) => role.id === action?.id)
  return (
    <Stack>
      <Group justify="space-between">
        <Title order={1}>{t("orgRolesTitle")}</Title>
        <Group>
          <TableColumnControls columns={columns} disabled={query.isFetching || !readable} />
          {allow("team.roles.create") && (
            <Button
              disabled={query.isFetching || !readable}
              onClick={() => setAction({ kind: "create" })}
            >
              {t("teamCreateRole")}
            </Button>
          )}
        </Group>
      </Group>
      <ListSurface
        label={t("orgRoleTable")}
        footer={
          readable &&
          query.data && (
            <ManagementPagination
              state={list.state}
              total={query.data.total}
              change={list.change}
              disabled={query.isFetching || !readable}
            />
          )
        }
      >
        <ModuleState
          loading={query.isPending}
          refreshing={query.isFetching && !query.isPending}
          error={query.error}
          empty={false}
          retry={() => {
            void query.refetch()
          }}
        />
        {readable && query.data && (
          <ListTable
            responsiveColumns={columns.columns
              .filter((column) => columns.visible(column.id))
              .map((column) => ({ label: column.label, width: columns.width(column.id) }))}
          >
            <Table.Thead role="rowgroup">
              <Table.Tr role="row">
                <TableColumnHeader
                  key={tableScope + "role"}
                  id="role"
                  columns={columns}
                  disabled={query.isFetching || !readable}
                >
                  {t("orgRole")}
                </TableColumnHeader>
                {columns.visible("permissions") && (
                  <TableColumnHeader
                    key={tableScope + "permissions"}
                    id="permissions"
                    columns={columns}
                    disabled={query.isFetching || !readable}
                  >
                    {t("orgPermissions")}
                  </TableColumnHeader>
                )}
                {columns.visible("type") && (
                  <TableColumnHeader
                    key={tableScope + "type"}
                    id="type"
                    columns={columns}
                    disabled={query.isFetching || !readable}
                  >
                    {t("orgType")}
                  </TableColumnHeader>
                )}
                <TableColumnHeader
                  key={tableScope + "actions"}
                  id="actions"
                  columns={columns}
                  disabled={query.isFetching || !readable}
                >
                  {t("orgActions")}
                </TableColumnHeader>
              </Table.Tr>
            </Table.Thead>
            <Table.Tbody role="rowgroup">
              {query.data.items.map((role) => (
                <Table.Tr role="row" key={"role:" + role.id}>
                  <ListCell rowHeader label={t("orgRole")} className={classes.name}>
                    {role.protected
                      ? t(role.code === "owner" ? "organizationOwner" : "orgBaseRole")
                      : role.name}
                    <Text size="xs" c="dimmed">
                      {role.code}
                    </Text>
                  </ListCell>
                  {columns.visible("permissions") && (
                    <ListCell label={t("orgPermissions")}>
                      {t("orgPermissionCount", { count: role.permissions.length })}
                    </ListCell>
                  )}
                  {columns.visible("type") && (
                    <ListCell label={t("orgType")}>
                      <Badge color="gray" variant="light">
                        {t(role.protected ? "teamSystemRole" : "orgCustomRole")}
                      </Badge>
                    </ListCell>
                  )}
                  <ListCell label={t("orgActions")}>
                    <Group gap="xs">
                      <Button
                        variant="subtle"
                        size="sm"
                        disabled={query.isFetching || !readable}
                        onClick={() => setAction({ kind: "permissions", id: role.id })}
                      >
                        {t(
                          allow("team.roles.permissions.update") && role.canManage
                            ? "orgConfigurePermissions"
                            : "orgViewPermissions"
                        )}
                      </Button>
                      {allow("team.roles.delete") && role.canManage && (
                        <Button
                          variant="subtle"
                          color="red"
                          size="sm"
                          disabled={query.isFetching || !readable}
                          onClick={() => setAction({ kind: "delete", id: role.id })}
                        >
                          {t("teamDeleteRole")}
                        </Button>
                      )}
                    </Group>
                  </ListCell>
                </Table.Tr>
              ))}
              {!query.data.items.length && (
                <Table.Tr role="row">
                  <Table.Td
                    colSpan={columns.columns.filter((column) => columns.visible(column.id)).length}
                  >
                    <Text role="status" p="xl" ta="center" c="dimmed">
                      {t("applicationEmpty")}
                    </Text>
                  </Table.Td>
                </Table.Tr>
              )}
            </Table.Tbody>
          </ListTable>
        )}
      </ListSurface>
      {readable && action && owner && (action.kind === "create" || target) && (
        <RoleDialog
          key={ownerKey + action.kind + action.id}
          kind={action.kind}
          role={target}
          close={() => setAction(null)}
        />
      )}
    </Stack>
  )
}
