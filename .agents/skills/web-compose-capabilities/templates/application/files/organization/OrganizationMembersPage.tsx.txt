import { ApiError } from "../../data/api/client/httpClient"
import { ListSurface, ListTable, ListCell } from "../../pagination/ListSurface"
import {
  useTableColumns,
  TableColumnControls,
  TableColumnHeader
} from "../../pagination/TableColumns"
import { useEffect, useState } from "react"
import { useQuery } from "@tanstack/react-query"
import { Badge, Button, Group, Stack, Table, Text, Title } from "@mantine/core"
import { useTranslation } from "react-i18next"
import { ModuleState } from "../application/ModuleState"
import { useManagementScope, useManagementRejection } from "./useManagement"
import { useManagementList } from "./managementList"
import { membersQuery } from "./managementQueries"
import { MemberDialog, type MemberAction } from "./MemberDialog"
import { ManagementPagination } from "./ManagementParts"
import classes from "./management.module.css"
import "../../i18n/organizationManagement"

/** 当前组织成员列表按服务端稳定 ID 排序，行操作同时受对象和精确权限约束。 */
export function OrganizationMembersPage() {
  const { t, i18n } = useTranslation(),
    { owner, ownerKey, allow } = useManagementScope()
  const list = useManagementList(ownerKey)
  const tableScope = JSON.stringify([owner?.accountId, owner?.organizationId])
  const columns = useTableColumns(
    "organization-members",
    tableScope,
    [
      { id: "member", label: t("orgMember"), required: true, width: 240 },
      { id: "roles", label: t("orgRoles"), width: 200 },
      { id: "actions", label: t("orgActions"), required: true, width: 280 }
    ],
    true
  )
  const [action, setAction] = useState<{ kind: MemberAction; id?: string } | null>(null)
  const query = useQuery({
    ...membersQuery(ownerKey, owner?.organizationId ?? "", i18n.language, list.state),
    enabled: !!owner
  })
  const denied =
    query.error instanceof ApiError && (query.error.status === 401 || query.error.status === 403)
  const readable = !!owner && !denied
  useEffect(() => {
    if (!readable) setAction(null)
  }, [readable])
  useManagementRejection(query.error, ownerKey)
  useEffect(() => {
    if (query.isSuccess && !query.isFetching && !query.isPlaceholderData)
      list.correct(query.data.total)
  }, [
    query.data,
    query.isFetching,
    query.isSuccess,
    ownerKey,
    list.state.page,
    list.state.pageSize
  ])
  const target = query.data?.items.find((row) => row.id === action?.id)
  return (
    <Stack>
      <Group justify="space-between">
        <Title order={1}>{t("orgMembersTitle")}</Title>
        <Group>
          <TableColumnControls columns={columns} disabled={query.isFetching || !readable} />
          {allow("team.members.invite") && (
            <Button
              disabled={query.isFetching || !readable}
              onClick={() => setAction({ kind: "invite" })}
            >
              {t("orgInviteMember")}
            </Button>
          )}
        </Group>
      </Group>
      <ListSurface
        label={t("orgMemberTable")}
        footer={
          readable &&
          query.data && (
            <ManagementPagination
              state={list.state}
              total={query.data.total}
              change={list.change}
              disabled={query.isFetching || !readable}
            />
          )
        }
      >
        <ModuleState
          loading={query.isPending}
          refreshing={query.isFetching && !query.isPending}
          error={query.error}
          empty={false}
          retry={() => {
            void query.refetch()
          }}
        />
        {readable && query.data && (
          <ListTable
            responsiveColumns={columns.columns
              .filter((column) => columns.visible(column.id))
              .map((column) => ({ label: column.label, width: columns.width(column.id) }))}
          >
            <Table.Thead role="rowgroup">
              <Table.Tr role="row">
                <TableColumnHeader
                  key={tableScope + "member"}
                  id="member"
                  columns={columns}
                  disabled={query.isFetching || !readable}
                >
                  {t("orgMember")}
                </TableColumnHeader>
                {columns.visible("roles") && (
                  <TableColumnHeader
                    key={tableScope + "roles"}
                    id="roles"
                    columns={columns}
                    disabled={query.isFetching || !readable}
                  >
                    {t("orgRoles")}
                  </TableColumnHeader>
                )}
                <TableColumnHeader
                  key={tableScope + "actions"}
                  id="actions"
                  columns={columns}
                  disabled={query.isFetching || !readable}
                >
                  {t("orgActions")}
                </TableColumnHeader>
              </Table.Tr>
            </Table.Thead>
            <Table.Tbody role="rowgroup">
              {query.data.items.map((member) => (
                <Table.Tr role="row" key={"member:" + member.id}>
                  <ListCell rowHeader label={t("orgMember")} className={classes.name}>
                    {member.displayName}
                    {member.accountId === owner?.accountId && (
                      <Text size="xs" c="dimmed">
                        {t("orgMe")}
                      </Text>
                    )}
                  </ListCell>
                  {columns.visible("roles") && (
                    <ListCell label={t("orgRoles")}>
                      <Group gap="xs">
                        {member.roleNames.map((name, index) => (
                          <Badge key={member.roleIds[index] ?? name} color="gray" variant="light">
                            {name === "Owner"
                              ? t("organizationOwner")
                              : name === "Member"
                                ? t("orgBaseRole")
                                : name}
                          </Badge>
                        ))}
                      </Group>
                    </ListCell>
                  )}
                  <ListCell label={t("orgActions")}>
                    <Group gap="xs">
                      {allow("team.members.nickname.update") && (
                        <Button
                          size="sm"
                          variant="subtle"
                          disabled={query.isFetching || !readable}
                          onClick={() => setAction({ kind: "nickname", id: member.id })}
                        >
                          {t("orgNicknameAction")}
                        </Button>
                      )}
                      {allow("team.members.roles.assign") &&
                        member.canAssignRoles &&
                        member.accountId !== owner?.accountId && (
                          <Button
                            size="sm"
                            variant="subtle"
                            disabled={query.isFetching || !readable}
                            onClick={() => setAction({ kind: "roles", id: member.id })}
                          >
                            {t("orgAssignRoles")}
                          </Button>
                        )}
                      {allow("team.members.remove") &&
                        member.canRemove &&
                        member.accountId !== owner?.accountId && (
                          <Button
                            size="sm"
                            variant="subtle"
                            color="red"
                            disabled={query.isFetching || !readable}
                            onClick={() => setAction({ kind: "remove", id: member.id })}
                          >
                            {t("teamRemove")}
                          </Button>
                        )}
                    </Group>
                  </ListCell>
                </Table.Tr>
              ))}
              {!query.data.items.length && (
                <Table.Tr role="row">
                  <Table.Td
                    colSpan={columns.columns.filter((column) => columns.visible(column.id)).length}
                  >
                    <Text role="status" p="xl" ta="center" c="dimmed">
                      {t("applicationEmpty")}
                    </Text>
                  </Table.Td>
                </Table.Tr>
              )}
            </Table.Tbody>
          </ListTable>
        )}
      </ListSurface>
      {readable && action && owner && (action.kind === "invite" || target) && (
        <MemberDialog
          key={ownerKey + action.kind + action.id}
          kind={action.kind}
          member={target}
          close={() => setAction(null)}
        />
      )}
    </Stack>
  )
}
