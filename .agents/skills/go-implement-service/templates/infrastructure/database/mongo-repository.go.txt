// Package repository 提供强制组织隔离与乐观版本的 MongoDB 文档仓储。
package repository

import (
	"context"
	"errors"
	"go.mongodb.org/mongo-driver/v2/bson"
	"go.mongodb.org/mongo-driver/v2/mongo"
)

// ErrConflict 表示文档不存在、组织不匹配或版本已变化，不泄漏其他组织的数据存在性。
var ErrConflict = errors.New("repository: 文档不存在或版本冲突")

// Document 把作用域、版本与业务载荷分开，调用方不能覆盖归属字段。
type Document[T any] struct {
	ID             string `bson:"_id"`
	OrganizationID string `bson:"organization_id"`
	Version        int64  `bson:"version"`
	Value          T      `bson:"value"`
}

// Repository 持有服务端确定的集合，不接受客户端动态集合名称。
type Repository[T any] struct{ collection *mongo.Collection }

// New 创建集合访问器，不自动创建索引或修改已有数据。
func New[T any](db *mongo.Database, collection string) *Repository[T] {
	return &Repository[T]{collection: db.Collection(collection)}
}

// scopedFilter 要求调用方传入已授权组织，并将版本条件与归属取交集。
func scopedFilter(scope, id string, version int64) (bson.D, error) {
	if scope == "" || id == "" || version < 0 {
		return nil, errors.New("repository: 作用域、标识或版本无效")
	}
	filter := bson.D{{Key: "_id", Value: id}, {Key: "organization_id", Value: scope}}
	if version > 0 {
		filter = append(filter, bson.E{Key: "version", Value: version})
	}
	return filter, nil
}

// Insert 插入带作用域和初始版本的真实文档。
func (r *Repository[T]) Insert(ctx context.Context, scope, id string, value T) error {
	if _, err := scopedFilter(scope, id, 0); err != nil {
		return err
	}
	_, err := r.collection.InsertOne(ctx, Document[T]{ID: id, OrganizationID: scope, Version: 1, Value: value})
	return err
}

// FindByID 在数据库查询中强制组织过滤。
func (r *Repository[T]) FindByID(ctx context.Context, scope, id string) (Document[T], error) {
	var result Document[T]
	filter, err := scopedFilter(scope, id, 0)
	if err != nil {
		return result, err
	}
	err = r.collection.FindOne(ctx, filter).Decode(&result)
	if errors.Is(err, mongo.ErrNoDocuments) {
		err = ErrConflict
	}
	return result, err
}

// Update 只在同组织且版本匹配时原子替换载荷并递增版本。
func (r *Repository[T]) Update(ctx context.Context, scope, id string, version int64, value T) error {
	if version < 1 {
		return ErrConflict
	}
	filter, err := scopedFilter(scope, id, version)
	if err != nil {
		return err
	}
	result, err := r.collection.UpdateOne(ctx, filter, bson.D{{Key: "$set", Value: bson.D{{Key: "value", Value: value}}}, {Key: "$inc", Value: bson.D{{Key: "version", Value: 1}}}})
	if err != nil {
		return err
	}
	if result.MatchedCount != 1 {
		return ErrConflict
	}
	return nil
}

// Delete 只删除当前组织且版本匹配的文档。
func (r *Repository[T]) Delete(ctx context.Context, scope, id string, version int64) error {
	if version < 1 {
		return ErrConflict
	}
	filter, err := scopedFilter(scope, id, version)
	if err != nil {
		return err
	}
	result, err := r.collection.DeleteOne(ctx, filter)
	if err != nil {
		return err
	}
	if result.DeletedCount != 1 {
		return ErrConflict
	}
	return nil
}
