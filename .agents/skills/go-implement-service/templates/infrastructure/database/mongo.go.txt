// Package database 提供 MongoDB 专用连接，禁止退回 GORM。
package database

import (
	"context"
	"errors"
	"go.mongodb.org/mongo-driver/v2/mongo"
	"go.mongodb.org/mongo-driver/v2/mongo/options"
	"go.mongodb.org/mongo-driver/v2/mongo/readpref"
	"project_id/internal/config"
	"time"
)

// Open 在有限预算内连接并验证主节点，不执行隐式建库或迁移。
func Open(cfg config.Database) (*mongo.Database, error) {
	client, err := mongo.Connect(options.Client().ApplyURI(cfg.DSN).SetConnectTimeout(10 * time.Second).SetServerSelectionTimeout(10 * time.Second).SetMaxPoolSize(100))
	if err != nil {
		return nil, errors.New("database: MongoDB 配置无效")
	}
	ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
	defer cancel()
	if err := client.Ping(ctx, readpref.Primary()); err != nil {
		_ = client.Disconnect(ctx)
		return nil, errors.New("database: MongoDB 连接失败")
	}
	return client.Database(cfg.Name), nil
}

// Close 在有限预算内释放 MongoDB 连接池。
func Close(db *mongo.Database) {
	if db == nil {
		return
	}
	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Second)
	defer cancel()
	_ = db.Client().Disconnect(ctx)
}
